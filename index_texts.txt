document.addEventListener('DOMContentLoaded', () => {
            const html = document.documentElement;

            // Drawer Control
            const menuDrawer = document.getElementById('mobile-menu-drawer');
            const menuContent = document.getElementById('drawer-content');
            const menuBackdrop = document.getElementById('drawer-backdrop');

            const openDrawer = () => {
                menuDrawer.classList.remove('pointer-events-none');
                menuContent.classList.remove('translate-y-[-100%]');
                menuBackdrop.classList.add('opacity-50', 'pointer-events-auto');
                document.body.style.overflow = 'hidden';
            };
            const closeDrawer = () => {
                menuContent.classList.add('translate-y-[-100%]');
                menuBackdrop.classList.remove('opacity-50', 'pointer-events-auto');
                setTimeout(() => { menuDrawer.classList.add('pointer-events-none'); document.body.style.overflow = ''; }, 300);
            };

            document.getElementById('mobile-menu-open').addEventListener('click', openDrawer);
            document.getElementById('mobile-menu-close').addEventListener('click', closeDrawer);
            menuBackdrop.addEventListener('click', closeDrawer);

            // Accordions
            const setupAccordion = (toggleId, menuId, iconId) => {
                const toggle = document.getElementById(toggleId);
                toggle?.addEventListener('click', () => {
                    const menu = document.getElementById(menuId);
                    const icon = document.getElementById(iconId);
                    menu.classList.toggle('hidden');
                    icon.classList.toggle('rotate-180');
                });
            };
            setupAccordion('mobile-homes-toggle', 'mobile-homes-menu', 'homes-icon');
            setupAccordion('mobile-dashboard-toggle', 'mobile-dashboard-menu', 'dashboard-icon');

            // Theme Switcher
            const themeToggles = [
                document.getElementById('theme-toggle'), 
                document.getElementById('drawer-theme-toggle'),
                document.getElementById('mobile-theme-toggle-quick')
            ];
            
            const applyTheme = (theme) => {
                html.setAttribute('data-theme', theme);
                html.classList.toggle('dark', theme === 'dark');
                document.querySelectorAll('.light-icon').forEach(i => i.classList.toggle('hidden', theme === 'dark'));
                document.querySelectorAll('.dark-icon').forEach(i => i.classList.toggle('hidden', theme === 'light'));
            };
            
            themeToggles.forEach(btn => {
                btn?.addEventListener('click', () => {
                    const next = html.getAttribute('data-theme') === 'light' ? 'dark' : 'light';
                    localStorage.setItem('theme', next);
                    applyTheme(next);
                });
            });
            applyTheme(localStorage.getItem('theme') || 'light');

            // Language RTL toggle
            const langToggles = [document.getElementById('lang-icon'), document.getElementById('mobile-lang-icon-drawer')];
            langToggles.forEach(btn => btn?.addEventListener('click', () => {
                const current = localStorage.getItem('language') === 'ar' ? 'en' : 'ar';
                localStorage.setItem('language', current);
                html.setAttribute('dir', current === 'ar' ? 'rtl' : 'ltr');
            }));
            html.setAttribute('dir', localStorage.getItem('language') === 'ar' ? 'rtl' : 'ltr');

            // Active Nav Styling
            const setActiveNavLink = () => {
                const currentPath = window.location.pathname.split('/').pop() || 'index.html';
                
                // Desktop Links
                document.querySelectorAll('.nav-link').forEach(link => {
                    if (link.getAttribute('href') === currentPath) {
                        link.classList.add('active', 'text-accent', 'font-bold');
                    } else {
                        link.classList.remove('active', 'text-accent', 'font-bold');
                    }
                });

                // Desktop Dropdowns active state
                if (['index.html', 'home2.html'].includes(currentPath)) {
                    document.getElementById('home-dropdown-btn')?.classList.add('active', 'text-accent', 'font-bold');
                }
                if (['admin.html', 'userdash.html'].includes(currentPath)) {
                    document.getElementById('dashboard-dropdown-btn')?.classList.add('active', 'text-accent', 'font-bold');
                }

                // Mobile Drawer links
                document.querySelectorAll('#drawer-content a').forEach(link => {
                    const href = link.getAttribute('href');
                    if (href === currentPath) {
                        link.classList.add('text-accent', 'font-bold');
                        link.classList.remove('text-primary-text', 'text-secondary-text');
                    } else {
                        if (link.closest('#mobile-homes-menu') || link.closest('#mobile-dashboard-menu')) {
                            link.classList.add('text-secondary-text');
                            link.classList.remove('text-accent', 'font-bold');
                        } else {
                            link.classList.add('text-primary-text');
                            link.classList.remove('text-accent', 'font-bold');
                        }
                    }
                });

                // Mobile Toggles active state
                if (['index.html', 'home2.html'].includes(currentPath)) {
                    document.getElementById('mobile-homes-toggle')?.classList.add('text-accent', 'font-bold');
                }
                if (['admin.html', 'userdash.html'].includes(currentPath)) {
                    document.getElementById('mobile-dashboard-toggle')?.classList.add('text-accent', 'font-bold');
                }
            };
            setActiveNavLink();

            // Inquiry pricing toggle
            const btnMonthly = document.getElementById("btn-monthly");
            const btnSeasonal = document.getElementById("btn-seasonal");
            const monthlyPlans = document.getElementById("monthly-plans");
            const seasonalPlans = document.getElementById("seasonal-plans");

            if (btnMonthly && btnSeasonal && monthlyPlans && seasonalPlans) {
                btnMonthly.addEventListener("click", () => {
                    btnMonthly.classList.add("bg-custom-gradient", "text-white", "shadow-md");
                    btnMonthly.classList.remove("text-primary-text", "hover:text-accent");

                    btnSeasonal.classList.remove("bg-custom-gradient", "text-white", "shadow-md");
                    btnSeasonal.classList.add("text-primary-text", "hover:text-accent");

                    monthlyPlans.classList.remove("hidden");
                    monthlyPlans.classList.add("grid");
                    seasonalPlans.classList.remove("grid");
                    seasonalPlans.classList.add("hidden");
                });

                btnSeasonal.addEventListener("click", () => {
                    btnSeasonal.classList.add("bg-custom-gradient", "text-white", "shadow-md");
                    btnSeasonal.classList.remove("text-primary-text", "hover:text-accent");

                    btnMonthly.classList.remove("bg-custom-gradient", "text-white", "shadow-md");
                    btnMonthly.classList.add("text-primary-text", "hover:text-accent");

                    seasonalPlans.classList.remove("hidden");
                    seasonalPlans.classList.add("grid");
                    monthlyPlans.classList.remove("grid");
                    monthlyPlans.classList.add("hidden");
                });
            }
        
            // Desktop dropdown click support for touch devices
            document.querySelectorAll('.dropdown-wrapper').forEach(wrapper => {
                const btn = wrapper.querySelector('button');
                const menu = wrapper.querySelector('.dropdown-menu');
                if (btn && menu) {
                    btn.addEventListener('click', (e) => {
                        if (window.innerWidth >= 1024) {
                            e.stopPropagation();
                            const isVisible = menu.style.opacity === '1';
                            // Reset other menus
                            document.querySelectorAll('.dropdown-menu').forEach(m => {
                                m.style.opacity = '';
                                m.style.visibility = '';
                                m.style.transform = '';
                            });
                            if (!isVisible) {
                                menu.style.opacity = '1';
                                menu.style.visibility = 'visible';
                                menu.style.transform = 'translateY(0)';
                            }
                        }
                    });
                }
            });
            document.addEventListener('click', () => {
                document.querySelectorAll('.dropdown-menu').forEach(m => {
                    m.style.opacity = '';
                    m.style.visibility = '';
                    m.style.transform = '';
                });
            });
});
